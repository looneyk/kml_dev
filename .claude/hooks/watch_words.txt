# Describing a function by another function instead of saying what it does
re:\bdoes what [\w:.()]+ does\b
comment-re:\bas (?!(?:they|it|we|you|he|she|this|that|these|those|there|one|each|all)\b)[\w:.()']+(?: [\w:.()']+)? (?:does|do|did)\b(?! not\b)

# A possessive "own" with no noun after it: "the game's own", "the layout snaps them to its own"
re:(?<!of )\b(?:[\w`]+'s?|its|their) own(?=\s*(?:[,.;:)]|$))

# Clipped openings, at the start of a sentence, a list item, a table cell or a comment, in documents as well as code.
# Case-sensitive, so a sentence wrapped onto a new line partway through does not match
re:(?-i:(?:^|[.!?]\s+)[\s>*|\-]*(?:\d+\.\s+)?(?:\*\*)?(?:What an?|Whether)\b)
re:(?-i:(?:^|[.!?]\s+)[\s>*|\-]*(?:\d+\.\s+)?(?:\*\*)?On an? [^:.]*:)
re:(?-i:(?:^|[.!?]\s+)[\s>*|\-]*(?:\d+\.\s+)?(?:\*\*)?This (?:plan|document|doc|file|section|page|reference)\b)

# A heading or a bold label that opens with a question word and is not a question; it is a noun phrase instead
re:(?-i:^#+\s+(?:Part \d+:\s+)?(?:What|Whether|How|Where|Which|Why|Who)\b(?!.*\?\s*$))
re:(?-i:\*\*(?:What|Whether|How|Where|Which|Why|Who)\b[^*?]*[^?*]\*\*)
